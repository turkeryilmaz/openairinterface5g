add_library(log_headers INTERFACE)
target_include_directories(log_headers INTERFACE .)
target_link_libraries(log_headers INTERFACE T_headers)

set(log_sources log.c flight_recorder.c flight_monitor.c)
if (ENABLE_LTTNG)
  find_package(LTTngUST 2.3.8 EXACT REQUIRED)
  set(log_sources ${log_sources} lttng-tp.c)
else()
  message(STATUS "LTTNG support disabled")
endif()
add_library(LOG ${log_sources})
target_link_libraries(LOG PRIVATE CONFIG_LIB)
target_link_libraries(LOG PUBLIC log_headers)
if (ENABLE_LTTNG)
  target_link_libraries(LOG PUBLIC lttng-ust)
endif()

# The optional recorder is inert on other OAI-supported architectures.
if (ENABLE_TESTS AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|aarch64)$")
  add_executable(flight_recorder_test flight_recorder.c flight_tests/flight_recorder_test.c)
  target_include_directories(flight_recorder_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_definitions(flight_recorder_test PRIVATE FLIGHT_RECORDER_TESTING)
  target_link_libraries(flight_recorder_test PRIVATE pthread)
  file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/flight_test_output)
  add_test(NAME flight_recorder COMMAND flight_recorder_test ${CMAKE_CURRENT_BINARY_DIR}/flight_test_output)
  set_tests_properties(flight_recorder PROPERTIES TIMEOUT 30)

  add_executable(flight_monitor_test flight_monitor.c flight_tests/flight_monitor_test.c)
  target_include_directories(flight_monitor_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_definitions(flight_monitor_test PRIVATE FLIGHT_MONITOR_TESTING)
  target_link_libraries(flight_monitor_test PRIVATE pthread)
  add_test(NAME flight_monitor COMMAND flight_monitor_test)
  set_tests_properties(flight_monitor PROPERTIES TIMEOUT 30)
endif()
